# LLM providers.

anthropic==0.109.1
httpx==0.28.1

# Encryption for LLM provider API keys stored in the database (Fernet symmetric

cryptography==49.0.0

# Observability (MUS-25). OpenTelemetry GenAI semantic conventions: the API is
# imported unconditionally, the SDK only installs a provider when
# OTEL_EXPORTER_OTLP_ENDPOINT is set (see project/app/services/telemetry/).
#
# The OTLP exporter is the HTTP one, NOT opentelemetry-exporter-otlp-proto-grpc:
# gRPC drags in `grpcio`, whose wheels lag new Python minors and whose source
# build is the most common dependency failure in a CI matrix -- and this repo's
# matrix includes 3.13. Phoenix serves OTLP/HTTP on the same port as its UI, so
# nothing is given up (docker-compose.yml).
opentelemetry-api==1.44.0
opentelemetry-sdk==1.44.0
opentelemetry-semantic-conventions==0.65b0     # released in lockstep with api 1.44.0
opentelemetry-exporter-otlp-proto-http==1.44.0
